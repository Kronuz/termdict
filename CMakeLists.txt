cmake_minimum_required(VERSION 3.14)
project(termdict LANGUAGES CXX)

include(FetchContent)
FetchContent_Declare(fst GIT_REPOSITORY https://github.com/Kronuz/fst.git GIT_TAG 24da59a9e3d7225865335dcdc71ddeb2e1cb66ec)
FetchContent_Declare(varint GIT_REPOSITORY https://github.com/Kronuz/varint.git GIT_TAG 21af82a64fd5860d14d5ec9591bec2ffd5d662d1)
FetchContent_MakeAvailable(fst varint)

# Header-only: termdict.hh is the whole implementation.
add_library(termdict INTERFACE)
target_include_directories(termdict INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(termdict INTERFACE cxx_std_20)
target_link_libraries(termdict INTERFACE fst::fst varint::varint)
add_library(termdict::termdict ALIAS termdict)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(termdict_test test/test.cc)
  target_link_libraries(termdict_test PRIVATE termdict)
  target_compile_definitions(termdict_test PRIVATE TERMDICT_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
  add_test(NAME termdict COMMAND termdict_test)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_executable(termdict_bench bench/bench.cc)
  target_link_libraries(termdict_bench PRIVATE termdict)
  option(TERMDICT_FUZZ "Build the libFuzzer target" OFF)
  if(TERMDICT_FUZZ)
    add_executable(termdict_fuzz fuzz/fuzz_reader.cc)
    target_link_libraries(termdict_fuzz PRIVATE termdict)
    target_compile_options(termdict_fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-sanitize-recover=all)
    target_link_options(termdict_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  endif()
endif()
